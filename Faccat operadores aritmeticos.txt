algoritmo "E08"

var
   totalEleitores, votosBrancos, votosNulos, votosValidos: real

inicio

   escreva("Quantos eleitores no municipio: ")
   leia(totalEleitores)

   escreva("Quantos votos brancos tiveram? ")
   leia(votosBrancos)

   escreva("Quantos votos nulos tiveram? ")
   leia(votosNulos)

   escreva("Quantos votos validos tiveram? ")
   leia(votosValidos)

   escreval("O porcentual dos brancos são: ", (votosBrancos / totalEleitores) * 100, "%")
   escreval("O porcentual dos nulos são: ", (votosNulos / totalEleitores) * 100, "%")
   escreval("O porcentual dos votos validos são: ", (votosValidos / totalEleitores) * 100, "%")

fimalgoritmo

algoritmo "E11"

var
   carrosVendidos, valorVendas, salarioFixo, valorPorCarro: real

inicio

   escreva("Quantos carros foram vendidos? ")
   leia(carrosVendidos)

   escreva("Qual o valor total das vendas? ")
   leia(valorVendas)

   escreva("Qual o salario fixo? ")
   leia(salarioFixo)

   escreva("Quanto recebe por carro vendido? ")
   leia(valorPorCarro)

   escreval("O salario final do vendedor é: ", salarioFixo + (carrosVendidos * valorPorCarro) + (valorVendas * 5 / 100))
